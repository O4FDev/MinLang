cmake_minimum_required(VERSION 3.22)
project(minyar_hello C)
if(NOT DEFINED MINYAR_ARCHIVE OR NOT EXISTS "${MINYAR_ARCHIVE}")
    message(FATAL_ERROR "Set MINYAR_ARCHIVE to the matching arm64-v8a or x86_64 Minyar archive")
endif()
add_library(minyar_core STATIC IMPORTED)
set_target_properties(minyar_core PROPERTIES IMPORTED_LOCATION "${MINYAR_ARCHIVE}")
add_library(minyar_hello SHARED minyar-jni.c)
target_link_libraries(minyar_hello PRIVATE minyar_core m)
target_link_options(minyar_hello PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,defs" "-Wl,-z,max-page-size=16384")
