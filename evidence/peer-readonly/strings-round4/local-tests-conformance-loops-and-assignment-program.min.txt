record Player { name: Text; x: Float; health: Integer; inventory: List<Integer> }
record Chunk { blocks: List<Integer> }
record World { chunks: List<Chunk>; player: Player }

function hurt(player: Player, amount: Integer) {
    player.health -= amount
}

let total = 0
for i in 0..10 {
    if i == 3 { continue }
    if i == 7 { break }
    total += i
}
print(total)

let words = ["alpha", "beta", "gamma"]
for word in words {
    print(word)
}
for c in "héllo" {
    if c == 'l' { continue }
    print(c)
}

let p = Player { name: "Steve"; x: 1.5; health: 20; inventory: [] }
p.x += 2.25
p.name += " the Miner"
hurt(p, 5)
p.inventory = [1, 2, 3]
p.inventory[1] *= 10
print(p.x)
print(p.name)
print(p.health)
print(p.inventory[1])

let blocks: List<Integer> = []
for i in 0..8 { blocks.add(0) }
let world = World { chunks: [Chunk { blocks: blocks }]; player: p }
world.chunks[0].blocks[5] = 42
world.player.health = 99
print(blocks[5])
print(p.health)

let n = 0
while true {
    n += 1
    if n % 2 == 0 { continue }
    if n > 9 { break }
}
print(n)

let grid: List<List<Integer>> = [[0, 0], [0, 0]]
grid[1][0] = 7
print(grid[1][0])

let count = 0
for row in 0..3 {
    for col in 0..3 {
        if col == row { break }
        count += 1
    }
}
print(count)
